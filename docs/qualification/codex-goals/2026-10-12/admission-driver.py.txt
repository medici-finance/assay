q=queue.Queue()
def reader(stream,label):
 for line in stream:q.put((label,line.rstrip()))
for stream,label in [(p.stdout,'RECV'),(p.stderr,'STDERR')]:threading.Thread(target=reader,args=(stream,label),daemon=True).start()
seq=0;thread=None;deadline=time.monotonic()+120

def send(method,params):
 global seq
 seq+=1;p.stdin.write(json.dumps({'jsonrpc':'2.0','id':seq,'method':method,'params':params})+'\n');p.stdin.flush();print('SEND',seq,method,json.dumps(params),flush=True);return seq

def recv():
 if time.monotonic()>deadline:raise TimeoutError('qualification bound reached')
 try:label,line=q.get(timeout=1)
 except queue.Empty:return {}
 print(label,line,flush=True)
 if label=='STDERR':return {}
 try:return json.loads(line)
 except ValueError:return {}

def wait_id(i, allow_error=False):
 while True:
  x=recv()
  if x.get('id')==i:
   if 'error' in x:
    if allow_error:return {'error':x['error']}
    raise RuntimeError(x['error'])
   return x['result']
try:

  except Exception as e:print('CLEANUP_RPC',repr(e),flush=True)
